# Dockerfile (già ottimizzato)
FROM golang:1.22-alpine AS builder
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 GOOS=linux go build -ldflags="-s -w" -o /dionisio ./cmd/dionisio

FROM alpine:3.19
RUN apk --no-cache add ca-certificates tzdata
WORKDIR /root
COPY --from=builder /dionisio .
EXPOSE 9090
USER nobody
HEALTHCHECK --interval=30s --timeout=5s --retries=3 CMD ["/dionisio", "-healthcheck"]
ENTRYPOINT ["/dionisio"]
